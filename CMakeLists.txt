if(JM_JUCE_PACKAGE_DIR)
 find_package(JUCE 8.0.6 EXACT CONFIG REQUIRED PATHS "${JM_JUCE_PACKAGE_DIR}" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
else()
 include(FetchContent)
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 51a8a6d7aeae7326956d747737ccf1575e61e209)
 FetchContent_MakeAvailable(JUCE)
endif()
# JUCE 8 defaults Windows peers to Direct2D before an editor can select GDI.
# Keep the renderer choice at peer creation so older drivers are never entered.
if(WIN32)
get_target_property(JM_GUI_MODULE juce::juce_gui_basics INTERFACE_JUCE_MODULE_PATH)
set(JM_WINDOWS_PEER "${JM_GUI_MODULE}/juce_gui_basics/native/juce_Windowing_windows.cpp")
file(READ "${JM_WINDOWS_PEER}" JM_PEER_SOURCE)
set(JM_ORIGINAL_PEER "return new HWNDComponentPeer { *this, styleFlags, (HWND) parentHWND, false, 1 };")
set(JM_SOFTWARE_PEER "return new HWNDComponentPeer { *this, styleFlags, (HWND) parentHWND, false, 0 }; // J Money software-renderer compatibility patch")
if(NOT JM_PEER_SOURCE MATCHES "J Money software-renderer compatibility patch")
 string(FIND "${JM_PEER_SOURCE}" "${JM_ORIGINAL_PEER}" JM_PEER_POSITION)
 if(JM_PEER_POSITION LESS 0)
  message(FATAL_ERROR "Unexpected JUCE Windows peer source; review the software renderer patch")
 endif()
 string(REPLACE "${JM_ORIGINAL_PEER}" "${JM_SOFTWARE_PEER}" JM_PEER_SOURCE "${JM_PEER_SOURCE}")
 file(WRITE "${JM_WINDOWS_PEER}" "${JM_PEER_SOURCE}")
endif()
# The pinned VBlank worker checked its exit flag only after a successful
# driver wait. Drivers/Wine that return failure could therefore deadlock
# stopThread(-1) during window/monitor lifecycle changes.
set(JM_VBLANK_FILE "${JM_GUI_MODULE}/juce_gui_basics/native/juce_VBlank_windows.cpp")
file(READ "${JM_VBLANK_FILE}" JM_VBLANK_SOURCE)
if(NOT JM_VBLANK_SOURCE MATCHES "J Money: honor shutdown on failed vblank")
 string(FIND "${JM_VBLANK_SOURCE}" "for (;;)" JM_VBLANK_POSITION)
 if(JM_VBLANK_POSITION LESS 0)
  message(FATAL_ERROR "Unexpected JUCE VBlank source; review the shutdown compatibility patch")
 endif()
 string(REPLACE "for (;;)" "while ((state.load() & flagExit) == 0) // J Money: honor shutdown on failed vblank" JM_VBLANK_SOURCE "${JM_VBLANK_SOURCE}")
 file(WRITE "${JM_VBLANK_FILE}" "${JM_VBLANK_SOURCE}")
endif()
endif()
